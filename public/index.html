function renderAiMsg(reply, perf, notes) { const box = document.getElementById('messages'); const d = document.createElement('div'); d.className = 'bubble ai-bubble'; let cleanText = reply; let choices = []; // 🛠️ Improved Regex to catch options even if AI makes minor formatting errors const match = reply.match(/\[A\.(.*)\|.*B\.(.*)\]/) || reply.match(/\[A:(.*)\|.*B:(.*)\]/); if(match) { cleanText = reply.split('[')[0]; choices = [match[1].trim(), match[2].trim()]; } else { // Fallback: If no options found, provide default ones to keep the flow choices = ["That's interesting!", "Tell me more!"]; } d.innerHTML = `
${cleanText}
`; if(perf) d.innerHTML += `
📊 ASSESSMENT: ${perf}
`; if(notes) d.innerHTML += `
${notes.replace(/\\n/g, '
')}
`; // 🔘 Render Buttons const cBox = document.createElement('div'); cBox.className = 'choice-box'; choices.forEach(c => { const b = document.createElement('button'); b.className = 'choice-btn'; b.innerText = c; b.onclick = () => sendToAI(c, 'click'); cBox.appendChild(b); }); d.appendChild(cBox); const hint = document.createElement('div'); hint.className = 'xp-hint'; hint.innerText = "🔥 SPEAK via Mic to earn 20 XP!"; d.appendChild(hint); box.appendChild(d); box.scrollTop = box.scrollHeight; window.speechSynthesis.speak(new SpeechSynthesisUtterance(cleanText)); }
