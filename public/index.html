// ... আগের CSS এবং HTML এলিমেন্ট অপরিবর্তিত থাকবে ... function renderMsg(txt, sender, review = "") { const box = document.getElementById('messages'); const d = document.createElement('div'); d.className = `bubble ${sender}-bubble`; let cleanTxt = txt; const match = txt.match(/\[(.*?)\]/); if(match) { cleanTxt = txt.replace(match[0], ""); const options = match[1].split('|'); d.innerHTML = `
${cleanTxt}
`; options.forEach(opt => { const btn = document.createElement('button'); btn.className = 'choice-btn'; btn.innerText = opt.trim(); btn.onclick = () => { renderMsg(opt, 'user'); callAPI({ message: opt, userId: UID, inputType: 'click' }); }; d.appendChild(btn); }); } else { d.innerText = txt; } // 🇧🇩 প্রমিত বাংলা রিভিউ ব্লক (ভয়েস হবে না) if(sender === 'ai' && review) { const rBox = document.createElement('div'); rBox.className = 'bn-review-box'; // এখানে প্রমিত বাংলার গুরুত্ব অনুযায়ী ফন্ট স্টাইল একটু উন্নত করা হয়েছে rBox.innerHTML = `
📋 প্রমিত বাংলা পর্যালোচনা:
${review}`; d.appendChild(rBox); } box.appendChild(d); box.scrollTop = box.scrollHeight; if(sender === 'ai') speak(cleanTxt); // কেবল ইংরেজি স্পোকেন অংশটি ভয়েস পড়বে }
